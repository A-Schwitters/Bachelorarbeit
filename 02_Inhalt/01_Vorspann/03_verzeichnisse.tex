% Alle definierten Einträge in die Verzeichnisse aufnehmen
\glsaddall

% Inhaltsverzeichnis
\iftoggle{showtoc}{
    \hypersetup{linkcolor=black}
    \tableofcontents
}{}

% Tabellenverzeichnis
\iftoggle{showtables}{
    \clearpage
    \phantomsection
    \addcontentsline{toc}{section}{Tabellenverzeichnis}
    \listoftables
}{}

% Abbildungsverzeichnis
\iftoggle{showfigures}{
    \clearpage
    \phantomsection
    \addcontentsline{toc}{section}{Abbildungsverzeichnis}
    \listoffigures
}{}

% Oszillogrammverzeichnis
\iftoggle{showoszillos}{
    \clearpage
    \phantomsection
    \addcontentsline{toc}{section}{Oszillogrammverzeichnis}
    \listofoszillo
}{}

% Diagrammverzeichnis
\iftoggle{showdiagrams}{
    \clearpage
    \phantomsection
    \addcontentsline{toc}{section}{Diagrammverzeichnis}
    \listofdiagram
}{}
